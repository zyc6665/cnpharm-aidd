```python
from app.tasks.executor import TaskExecutor
from app.models.tools import ToolVersion
from sqlalchemy import select

# 获取工具版本
result = await db.execute(
    select(ToolVersion).where(ToolVersion.id == task.tool_version_id)
)
tool_version = result.scalar_one_or_none()

if not tool_version:
    raise ValueError(f"工具版本不存在: {task.tool_version_id}")

# 创建执行器
executor = TaskExecutor(task, tool_version)

# 定义进度回调
async def progress_callback(progress: float, message: str):
    task.progress = progress
    task.progress_message = message
    await db.commit()

    await notify_task_log(task_id, "INFO", message)

# 执行任务
result = await executor.execute_with_progress(
    progress_callback=progress_callback
)

# 保存结果（executor已更新task状态，这里只保存output）
output_result = result.get("result", {})
```

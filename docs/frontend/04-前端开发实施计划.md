### 4.2 用户状态管理（增强）

```typescript
// stores/user.ts
import { defineStore } from 'pinia'
import { authApi } from '@/api/auth'

interface UserState {
  token: string | null
  refreshToken: string | null
  userInfo: UserInfo | null
  credits: number  // 新增:积分
  isGuest: boolean  // 新增:游客标识
}

interface UserInfo {
  id: string
  username: string
  email: string
  role: 'admin' | 'vip' | 'regular'
  memberExpireAt?: string  // 会员到期时间
  invitationCode: string  // 邀请码
}

export const useUserStore = defineStore('user', {
  state: (): UserState => ({
    token: null,
    refreshToken: null,
    userInfo: null,
    credits: 0,
    isGuest: false
  }),

  getters: {
    isLoggedIn: (state) => !!state.token && !state.isGuest,
    isAdmin: (state) => state.userInfo?.role === 'admin',
    isVIP: (state) => state.userInfo?.role === 'vip',
    hasEnoughCredits: (state) => (required: number) => state.credits >= required
  },

  actions: {
    async login(params: LoginParams) {
      const res = await authApi.login(params)
      this.token = res.data.access_token
      this.refreshToken = res.data.refresh_token
      this.userInfo = res.data.user
      this.credits = res.data.credits
      this.isGuest = false
    },

    async register(params: RegisterParams) {
      const res = await authApi.register(params)
      // 注册成功后自动登录
      this.token = res.data.access_token
      this.refreshToken = res.data.refresh_token
      this.userInfo = res.data.user
      this.credits = res.data.credits  // 注册奖励积分
      this.isGuest = false
    },

    // 新增:游客模式登录
    async guestLogin() {
      const res = await authApi.guestLogin()
      this.token = res.data.access_token
      this.isGuest = true
      this.credits = 0
    },

    // 新增:发送邮箱验证码
    async sendEmailCode(email: string) {
      await authApi.sendEmailCode(email)
    },

    // 新增:积分操作
    async fetchCredits() {
      const res = await authApi.getCredits()
      this.credits = res.data.credits
    },

    async consumeCredits(amount: number) {
      this.credits -= amount
    },

    async logout() {
      await authApi.logout()
      this.$reset()
    },

    async refreshAccessToken() {
      const res = await authApi.refreshToken(this.refreshToken!)
      this.token = res.data.access_token
      this.refreshToken = res.data.refresh_token
    }
  },

  persist: {
    paths: ['token', 'refreshToken', 'userInfo', 'credits', 'isGuest']
  }
})
```

```vue
<template>
  <AuthLayout>
    <div class="register-form">
      <h2>创建您的账号</h2>

          </a-checkbox>
        </a-form-item>

        <!-- 提交按钮 -->
        <a-form-item>
          <a-button
            type="primary"
            html-type="submit"

        </div>
      </a-form>

<script setup lang="ts">
import { reactive, ref } from 'vue'
import { message } from 'ant-design-vue'
import { useUserStore } from '@/stores/user'
import { useRouter } from 'vue-router'
import PasswordStrength from './components/PasswordStrength.vue'

const userStore = useUserStore()
const router = useRouter()
const loading = ref(false)
const countdown = ref(0)

const form = reactive({
  username: '',
  email: '',
  password: '',
  code: '',
  invitationCode: '',
  agree: false
})

const rules = {
  username: [
    { required: true, message: '请输入用户名' },
    { min: 3, max: 20, message: '用户名长度为3-20字符' },
    { pattern: /^[a-zA-Z0-9_\u4e00-\u9fa5]+$/, message: '仅支持字母数字下划线中文' }
  ],
  email: [
    { required: true, message: '请输入邮箱' },
    { type: 'email', message: '请输入有效的邮箱地址' }
  ],
  password: [
    { required: true, message: '请输入密码' },
    { min: 8, message: '密码至少8位' },
    { pattern: /^(?=.*[a-z])(?=.*[A-Z])(?=.*\d)/, message: '密码需包含大小写字母和数字' }
  ],
  code: [
    { required: true, message: '请输入验证码' },
    { len: 6, message: '验证码为6位' }
  ],
  agree: [
    {
      validator: (_, value) => value ? Promise.resolve() : Promise.reject('请阅读并同意服务条款'),
      message: '请阅读并同意服务条款'
    }
  ]
}

// 发送验证码
const sendCode = async () => {
  if (!form.email) {
    message.error('请先输入邮箱')
    return
  }

  try {
    await userStore.sendEmailCode(form.email)
    message.success('验证码已发送')
    countdown.value = 60
    const timer = setInterval(() => {
      countdown.value--
      if (countdown.value === 0) {
        clearInterval(timer)
      }
    }, 1000)
  } catch (error) {
    message.error('发送失败,请稍后重试')
  }
}

// 注册
const handleRegister = async () => {
  loading.value = true
  try {
    await userStore.register(form)
    message.success('注册成功!获得100积分奖励')
    router.push('/tools')
  } catch (error: any) {
    message.error(error.message || '注册失败')
  } finally {
    loading.value = false
  }
}
</script>
```

### 6.2 WebSocket 实现

```typescript
// composables/useWebSocket.ts
import { ref, onUnmounted } from 'vue'
import { useUserStore } from '@/stores/user'

export function useTaskWebSocket(taskId: string) {
  const userStore = useUserStore()
  const ws = ref<WebSocket | null>(null)
  const progress = ref(0)
  const status = ref('pending')
  const logs = ref<string[]>([])
  const isConnected = ref(false)

  const connect = () => {
    const wsUrl = `${import.meta.env.VITE_WS_URL}/ws/tasks/${taskId}?token=${userStore.token}`
    ws.value = new WebSocket(wsUrl)

    ws.value.onopen = () => {
      isConnected.value = true
    }

    ws.value.onmessage = (event) => {
      const data = JSON.parse(event.data)

      switch (data.type) {
        case 'progress':
          progress.value = data.progress
          break
        case 'status':
          status.value = data.status
          break
        case 'log':
          logs.value.push(data.message)
          break
        case 'completed':
          status.value = 'success'
          break
        case 'failed':
          status.value = 'failure'
          break
      }
    }

    ws.value.onclose = () => {
      isConnected.value = false
    }
  }

  const disconnect = () => {
    ws.value?.close()
  }

  const sendPing = () => {
    ws.value?.send(JSON.stringify({ action: 'ping' }))
  }

  onUnmounted(() => {
    disconnect()
  })

  return {
    connect,
    disconnect,
    sendPing,
    progress,
    status,
    logs,
    isConnected
  }
}
```

```vue
<template>
```

```vue
        :disabled="alreadySigned"
        :loading="loading"
        @click="handleSignIn"
      >

      </a-button>

      <div class="signin-rules">
        <p>签到规则：</p>
        <ul>
          <li>每日签到可获得积分奖励</li>
          <li>连续签到奖励递增（Day 7 双倍）</li>
          <li>中断后重新开始计算</li>
        </ul>
      </div>
    </div>
  </a-modal>
</template>

<script setup lang="ts">
import { ref, computed } from 'vue'
import { useUserStore } from '@/stores/user'
import { message } from 'ant-design-vue'

const props = defineProps<{
  visible: boolean
}>()

const emit = defineEmits<{
  (e: 'update:visible', value: boolean): void
}>()

const userStore = useUserStore()
const loading = ref(false)
const signedDays = ref(3) // 已签到天数
const alreadySigned = ref(false) // 今日是否已签到

const getReward = (day: number) => {
  const baseReward = 10
  if (day === 7) return baseReward * 2
  return baseReward + (day - 1) * 2
}

const todayReward = computed(() => {
  return getReward(signedDays.value + 1)
})

const handleSignIn = async () => {
  loading.value = true
  try {
    await userStore.dailySignIn()
    message.success(`签到成功!获得 ${todayReward.value} 积分`)
    signedDays.value++
    alreadySigned.value = true
  } catch (error: any) {
    message.error(error.message || '签到失败')
  } finally {
    loading.value = false
  }
}
</script>
```

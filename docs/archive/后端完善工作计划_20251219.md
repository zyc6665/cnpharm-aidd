            # ==================== 获取工具版本配置 ====================
            from app.models.tools import ToolVersion

            version_result = await db.execute(
                select(ToolVersion).where(ToolVersion.id == task.tool_version_id)
            )
            tool_version = version_result.scalar_one_or_none()

            if not tool_version:
                raise ValueError(f"工具版本不存在: {task.tool_version_id}")

            logger.info(f"开始执行工具: {tool_version.tool.name} v{tool_version.version}")

            # ==================== 创建执行器 ====================

            executor = TaskExecutor(task, tool_version)

            # ==================== 定义进度回调 ====================
            async def progress_callback(progress: float, message: str):

                # 更新数据库
                task.progress = progress
                task.progress_message = message
                await db.commit()

                # WebSocket通知
                await notify_task_progress(task_id, progress, message)
                await notify_task_log(task_id, "INFO", message)

                logger.debug(f"Task {task_id} progress: {progress:.1%} - {message}")

            result = await executor.execute_with_progress(
                progress_callback=progress_callback
            )

            logger.info(f"任务执行成功: {task_id}")

            # ==================== 保存结果 ====================
            # 注意：executor.execute_with_progress已经更新了task状态
            # 这里只需要保存output
            output_result = result.get("result", {})

            # 创建输出记录
            output = TaskOutput(
                task_id=task_id,
                result=output_result,
                output_type=result.get("output_type", "json"),
                file_ids=result.get("output_file_ids", [])
            )
            db.add(output)
            await db.commit()

            # WebSocket通知：任务完成
            await notify_task_log(task_id, "INFO", "任务执行成功")
            await notify_task_complete(task_id, output_result)

            return {
                "status": "success",
                "result": output_result
            }

        except Exception as e:
            # 错误处理
            logger.error(f"任务执行失败: {task_id}, 错误: {str(e)}", exc_info=True)

            # 更新失败状态
            task.status = TaskStatus.FAILURE.value
            task.finished_at = datetime.utcnow()
            task.error_message = str(e)
            task.progress_message = f"任务失败: {str(e)}"

            # 添加错误日志
            log = TaskLog(
                task_id=task_id,
                level="ERROR",
                message=f"任务执行失败: {str(e)}"
            )
            db.add(log)
            await db.commit()

            # WebSocket通知：任务失败
            await notify_task_log(task_id, "ERROR", f"任务执行失败: {str(e)}")
            await notify_task_error(task_id, str(e))

**替换为**:
```python
async def _download_input_files(self) -> List[str]:
    """下载输入文件到本地"""
    from app.storage.factory import get_storage
    from app.models.files import File

    file_paths = []
    work_dir = self._get_work_dir()
    storage = get_storage()

    if not self.task.inputs or not self.task.inputs.file_ids:
        return file_paths

    async with self.task.db_session() as db:
        for file_id in self.task.inputs.file_ids:
            try:
                # 从数据库获取文件信息
                result = await db.execute(
                    select(File).where(File.id == file_id)
                )
                file = result.scalar_one_or_none()

                if not file:
                    logger.warning(f"File {file_id} not found in database")
                    continue

                # 下载文件到本地
                local_path = os.path.join(work_dir, "input", file.original_filename)
                await storage.download(file.storage_path, local_path)
                file_paths.append(local_path)

                logger.info(f"Downloaded file: {file.original_filename}")

            except Exception as e:
                logger.error(f"Failed to download file {file_id}: {e}")
                raise TaskExecutionError(f"文件下载失败: {str(e)}")

    return file_paths
```

### 任务2.2：实现文件输出功能 ⏱️ 1小时

**文件**: `backend/app/tasks/executor.py`

**修改位置**: RemoteAPIExecutor._process_output_files (第177-184行)

**替换为**:
```python
async def _process_output_files(self, output_files: List[Dict]) -> List[str]:
    """处理输出文件"""
    from app.storage.factory import get_storage
    from app.models.files import File
    import uuid

    storage = get_storage()
    uploaded_ids = []

    async with self.task.db_session() as db:
        for file_info in output_files:
            try:
                # 假设file_info包含: {"url": "...", "filename": "..."}
                file_url = file_info.get("url")
                filename = file_info.get("filename", f"output_{uuid.uuid4().hex[:8]}")

                # 下载文件
                import httpx
                async with httpx.AsyncClient() as client:
                    response = await client.get(file_url)
                    response.raise_for_status()
                    content = response.content

                # 上传到存储
                file_id = str(uuid.uuid4())
                storage_path = f"task_outputs/{self.task.id}/{filename}"
                await storage.upload_bytes(content, storage_path)

                # 创建文件记录
                file = File(
                    id=file_id,
                    original_filename=filename,
                    storage_path=storage_path,
                    size_bytes=len(content),
                    content_type=file_info.get("content_type", "application/octet-stream"),
                    uploaded_by=self.task.user_id
                )
                db.add(file)
                uploaded_ids.append(file_id)

                logger.info(f"Uploaded output file: {filename}")

            except Exception as e:
                logger.error(f"Failed to process output file: {e}")

        await db.commit()

    return uploaded_ids
```

**Q1: 任务创建后仍然PENDING**
```bash
# 检查1: Celery Worker是否运行
docker ps | grep celery-worker

# 检查2: Redis是否可连接
docker exec -it scitool-redis redis-cli -a scitool123 ping

# 检查3: 队列中是否有任务
docker exec -it scitool-redis redis-cli -a scitool123 llen celery
```

**Q2: 执行器报错 "Module not found"**
```bash
# 确保在正确的Python环境
cd backend
pip install -r requirements.txt
```

**Q3: Docker执行器报错 "permission denied"**
```bash
# 检查Docker socket权限
ls -l /var/run/docker.sock

# 添加权限（如需要）
sudo chmod 666 /var/run/docker.sock
```
